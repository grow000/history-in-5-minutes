# Формат события (src/data/events/*.js)

Каждый файл: `export default [ {...}, {...} ]` — массив объектов:

{
  id: 'kulikovo-battle',            // латиница, kebab-case, уникальный
  title: 'Куликовская битва',
  era: 'medieval',                  // ancient | medieval | early-modern | modern | contemporary
  category: 'war',                  // war | politics | culture | science | religion | discovery | economy | society
  region: 'Русь',                   // короткое: Европа, Азия, Русь, Россия, Америка, Ближний Восток, Весь мир ...
  date: '8 сентября 1380 г.',       // для показа
  year: 1380,                       // число для сортировки (до н. э. — отрицательное)
  emoji: '⚔️',                      // один эмодзи-символ
  summary: '1–2 предложения для карточки (до ~200 символов).',
  whatHappened: 'Абзац 3–5 предложений: что произошло.',
  causes: ['Причина 1', 'Причина 2', 'Причина 3'],            // 3–5 пунктов
  course: ['Этап 1 ...', 'Этап 2 ...', 'Этап 3 ...'],        // 3–6 пунктов, по 1–3 предложения
  people: [{ name: 'Дмитрий Донской', role: 'великий князь московский, командовал русским войском' }], // 3–5
  timeline: [{ date: '1378', text: 'Битва на реке Воже' }],   // 4–7 пунктов по порядку
  facts: ['Факт 1', 'Факт 2', 'Факт 3', 'Факт 4', 'Факт 5'],  // РОВНО 5
  significance: 'Абзац 3–5 предложений: значение и последствия.',
  quiz: [                                                      // РОВНО 5
    { question: '...?', options: ['A', 'B', 'C', 'D'], answer: 2, explanation: '1 предложение пояснения.' }
  ]
}
